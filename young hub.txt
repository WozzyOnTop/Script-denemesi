local function resolveEmbeddedIcon(path, data, fallback)
    local decoder = rawget(Environment, "base64_decode") or rawget(Environment, "base64decode")
    local writer = rawget(Environment, "writefile")
    local customAsset = rawget(Environment, "getcustomasset") or rawget(Environment, "getsynasset")
    if type(customAsset) == "function" then
        local fileExists = rawget(Environment, "isfile")
        if type(fileExists) == "function" then
            local checked, exists = pcall(fileExists, path)
            if checked and exists then
                local loaded, asset = pcall(customAsset, path)
                if loaded and type(asset) == "string" and asset ~= "" then
                    return asset
                end
            end
        end
        if type(decoder) == "function" and type(writer) == "function" then
            local makeFolder = rawget(Environment, "makefolder")
            local isFolder = rawget(Environment, "isfolder")
            if type(makeFolder) == "function" then
                pcall(function()
                    if type(isFolder) ~= "function" or not isFolder("Young0xHub") then
                        makeFolder("Young0xHub")
                    end
                end)
            end
            local decoded, bytes = pcall(decoder, data)
            if decoded and type(bytes) == "string" then
                local written = pcall(writer, path, bytes)
                if written then
                    local loaded, asset = pcall(customAsset, path)
                    if loaded and type(asset) == "string" and asset ~= "" then
                        return asset
                    end
                end
            end
        end
    end
    return fallback
end

local function validFG100DeviceToken(value)
    return type(value) == "string" and #value >= 32 and #value <= 256
end

local function fg100LocalFingerprint(value)
    local first, second = 2166136261, 2246822519
    for index = 1, #value do
        local byte = string.byte(value, index)
        first = (first * 131 + byte) % 4294967296
        second = (second * 137 + byte + index) % 4294967296
    end
    return string.format("%08x%08x", first, second)
end

local function fg100DeviceKey(hwid)
    return tostring(player.UserId) .. ":" .. fg100LocalFingerprint(hwid)
end

local function fg100DeviceFile(hwid)
    return FG100_DEVICE_FOLDER
        .. "/fg100-device-"
        .. tostring(player.UserId)
        .. "-"
        .. fg100LocalFingerprint(hwid)
        .. ".txt"
end

local function loadFG100DeviceToken(hwid)
    local key = fg100DeviceKey(hwid)
    local cached = FG100_DEVICE_CACHE[key]
    if type(cached) == "table" and validFG100DeviceToken(cached.token) then
        return cached.token, cached.legacy == true
    end
    if type(isfile) ~= "function" or type(readfile) ~= "function" then
        return nil
    end
    local path = fg100DeviceFile(hwid)
    local okExists, exists = pcall(isfile, path)
    if okExists and exists then
        local okRead, value = pcall(readfile, path)
        if okRead and validFG100DeviceToken(value) then
            FG100_DEVICE_CACHE[key] = { token = value, legacy = false }
            return value, false
        end
    end
    okExists, exists = pcall(isfile, FG100_DEVICE_FILE_LEGACY)
    if not okExists or not exists then
        return nil
    end
    local okRead, value = pcall(readfile, FG100_DEVICE_FILE_LEGACY)
    if okRead and validFG100DeviceToken(value) then
        FG100_DEVICE_CACHE[key] = { token = value, legacy = true }
        return value, true
    end
    return nil
end

local function saveFG100DeviceToken(value, hwid)
    if not validFG100DeviceToken(value) then
        return false
    end
    local key = fg100DeviceKey(hwid)
    FG100_DEVICE_CACHE[key] = { token = value, legacy = false }
    if type(writefile) ~= "function" then
        return false
    end
    if type(makefolder) == "function" then
        pcall(makefolder, FG100_DEVICE_FOLDER)
    end
    return pcall(writefile, fg100DeviceFile(hwid), value)
end

local function clearFG100DeviceToken(hwid)
    FG100_DEVICE_CACHE[fg100DeviceKey(hwid)] = nil
    if type(isfile) ~= "function" or type(delfile) ~= "function" then
        return
    end
    local path = fg100DeviceFile(hwid)
    local okExists, exists = pcall(isfile, path)
    if okExists and exists then
        pcall(delfile, path)
    end
end

local function clearFG100LegacyDeviceToken()
    if type(isfile) ~= "function" or type(delfile) ~= "function" then
        return
    end
    local okExists, exists = pcall(isfile, FG100_DEVICE_FILE_LEGACY)
    if okExists and exists then
        pcall(delfile, FG100_DEVICE_FILE_LEGACY)
    end
end

local function rejectFG100DeviceToken(hwid, cameFromLegacy)
    clearFG100DeviceToken(hwid)
    if cameFromLegacy then
        clearFG100LegacyDeviceToken()
    end
end

local function validFG100Attempt(value)
    return type(value) == "string" and #value >= 32 and #value <= 128
        and value:match("^[%w_-]+$") ~= nil
end

local function fg100ActivationRequestId(hwid, activationCode)
    local key = fg100DeviceKey(hwid)
    local value = FG100_ATTEMPTS[key]
    if not validFG100Attempt(value) and type(readfile) == "function" then
        local ok, saved = pcall(readfile, fg100AttemptFile(hwid))
        if ok and validFG100Attempt(saved) then value = saved end
    end
    if not validFG100Attempt(value) then
        local material = tostring(player.UserId) .. ":" .. hwid .. ":" .. tostring(activationCode)
        value = "fg100-" .. fg100LocalFingerprint(material)
            .. "-" .. fg100LocalFingerprint(tostring(activationCode) .. ":" .. hwid)
    end
    assert(validFG100Attempt(value), "activation request unavailable")
    FG100_ATTEMPTS[key] = value
    if type(writefile) == "function" then
        if type(makefolder) == "function" then pcall(makefolder, FG100_DEVICE_FOLDER) end
        pcall(writefile, fg100AttemptFile(hwid), value)
    end
    return value
end

local function clearFG100Attempt(hwid)
    FG100_ATTEMPTS[fg100DeviceKey(hwid)] = nil
    if type(delfile) == "function" then pcall(delfile, fg100AttemptFile(hwid)) end
end

local function fg100ResponseStatus(response)
    local value = response and (response.StatusCode or response.Status or response.status_code)
    local direct = tonumber(value)
    if direct then
        return direct
    end
    local digits = tostring(value or ""):match("%d+")
    return digits and tonumber(digits) or 0
end

local function fg100ResponseBody(response)
    return response and (response.Body or response.body)
end

local FG100_RETRY_DELAYS = { 0.45, 0.9, 1.6 }
local FG100_REQUEST_TIMEOUT = 12

local function fg100TransientStatus(status)
    return status == -1
        or status == 0
        or status == 408
        or status == 425
        or status == 500
        or status == 502
        or status == 503
        or status == 504
        or (status >= 520 and status <= 524)
end

local function fg100Post(path, body, retryTransient)
    local encoded = HttpService:JSONEncode(body)
    local lastBody, lastStatus = nil, 0
    local totalAttempts = retryTransient and (#FG100_RETRY_DELAYS + 1) or 1
    for attempt = 1, totalAttempts do
        local completed, ok, response = false, false, nil
        task.spawn(function()
            ok, response = pcall(requestFn, {
                Url = FG100_API_BASE .. path,
                Method = "POST",
                Headers = {
                    ["content-type"] = "application/json",
                    ["cache-control"] = "no-store",
                },
                Body = encoded,
            })
            completed = true
        end)
        local deadline = os.clock() + FG100_REQUEST_TIMEOUT
        while not completed and os.clock() < deadline do
            task.wait(0.05)
        end
        if not completed then
            lastBody, lastStatus = nil, -1
        elseif ok and type(response) == "table" then
            lastBody = fg100ResponseBody(response)
            lastStatus = fg100ResponseStatus(response)
        else
            lastBody, lastStatus = nil, 0
        end

        if not fg100TransientStatus(lastStatus) or attempt >= totalAttempts then
            return lastBody, lastStatus
        end
        fg100LoadingNotice()
        task.wait(FG100_RETRY_DELAYS[attempt])
    end
    return lastBody, lastStatus
end

local function decodeFG100Response(raw)
    if type(raw) ~= "string" then
        return nil
    end
    local ok, decoded = pcall(HttpService.JSONDecode, HttpService, raw)
    return ok and type(decoded) == "table" and decoded or nil
end

local function validFG100Session(session, status)
    return status == 200 and type(session) == "table" and session.ok == true
        and type(session.sessionId) == "string" and #session.sessionId >= 20
        and type(session.sessionToken) == "string" and #session.sessionToken >= 32
end

local function fg100SessionFailure(session, status, hwid)
    local code = session and tostring(session.code or "") or ""
    if status == 403 and code == "device_activation_required" then
        clearFG100DeviceToken(hwid)
        if type(setFG100AccessVisual) == "function" then setFG100AccessVisual(false) end
        fg100Notice("Este dispositivo no está vinculado. Usá /fg100 registrar.")
    elseif status == 403 and code == "device_credential_invalid" then
        clearFG100DeviceToken(hwid)
        if type(setFG100AccessVisual) == "function" then setFG100AccessVisual(false) end
        fg100Notice("Este dispositivo necesita activarse de nuevo. Usá /fg100 registrar.")
    elseif status == 403 and code == "unauthorized" then
        Environment.Young0xFG100AccessDenied = true
        if type(setFG100AccessVisual) == "function" then setFG100AccessVisual(false) end
        fg100PaidAccessNotice()
    elseif status == 429 then
        Environment.Young0xFG100RetryAt = os.clock() + 8
        fg100LoadingNotice()
    elseif fg100TransientStatus(status) then
        Environment.Young0xFG100RetryAt = os.clock() + 4
        fg100LoadingNotice()
    else
        fg100Notice("No se pudo confirmar el acceso. Probá FG100 nuevamente")
    end
    return false
end

local function runFG100SecureRequest()
    if type(requestFn) ~= "function" then
        fg100Notice("Tu ejecutor no tiene soporte HTTP.")
        return false
    end
    if type(gethwid) ~= "function" then
        fg100Notice("Tu ejecutor no expone HWID. Usá uno compatible.")
        return false
    end

    local okHwid, hwid = pcall(gethwid)
    if not okHwid or type(hwid) ~= "string" or #hwid < 8 then
        fg100Notice("No se pudo leer el HWID de este ejecutor.")
        return false
    end

    local deviceToken, deviceTokenFromLegacy = loadFG100DeviceToken(hwid)
    local activationCode = Environment.Young0xFG100ActivationCode
    local activationUserId = Environment.Young0xFG100ActivationUserId
    if type(activationCode) == "string" and #activationCode >= 20
        and activationUserId ~= nil and tostring(activationUserId) ~= tostring(player.UserId) then
        fg100Notice("Este bloque es para Roblox ID " .. tostring(activationUserId)
            .. ". Entrá con esa cuenta o registrá la cuenta que estás usando.")
        return false
    end
    local session, sessionStatus
    local deviceNotSaved = false
    local function clearUsedCode()
        if Environment.Young0xFG100ActivationCode == activationCode then
            Environment.Young0xFG100ActivationCode = nil
            Environment.Young0xFG100ActivationUserId = nil
            clearFG100Attempt(hwid)
        end
    end
    local function requestSession()
        local raw, status = fg100Post("/v1/fg100/session", {
            robloxUserId = tostring(player.UserId),
            hwid = hwid,
            deviceToken = deviceToken or "",
        }, true)
        return decodeFG100Response(raw), status
    end

    if deviceToken then
        session, sessionStatus = requestSession()
        if validFG100Session(session, sessionStatus) then
            if deviceTokenFromLegacy and saveFG100DeviceToken(deviceToken, hwid) then
                clearFG100LegacyDeviceToken()
                deviceTokenFromLegacy = false
            end
        else
            local code = session and tostring(session.code or "") or ""
            if sessionStatus == 403 and (code == "device_activation_required" or code == "device_credential_invalid"
                or (code == "unauthorized" and type(activationCode) == "string" and #activationCode >= 20)) then
                if code ~= "unauthorized" then
                    rejectFG100DeviceToken(hwid, deviceTokenFromLegacy)
                    deviceTokenFromLegacy = false
                end
                deviceToken = nil
                if type(activationCode) ~= "string" or #activationCode < 20 then
                    return fg100SessionFailure(session, sessionStatus, hwid)
                end
            else
                return fg100SessionFailure(session, sessionStatus, hwid)
            end
        end
    end

    if not validFG100Session(session, sessionStatus)
        and type(activationCode) == "string" and #activationCode >= 20 then
        local activationRaw, activationStatus = fg100Post("/v1/fg100/activate", {
            robloxUserId = tostring(player.UserId),
            hwid = hwid,
            activationCode = activationCode,
            activationRequestId = fg100ActivationRequestId(hwid, activationCode),
        }, true)
        local activation = decodeFG100Response(activationRaw)
        if activationStatus ~= 200
            or not activation
            or activation.ok ~= true
            or not validFG100DeviceToken(activation.deviceToken) then
            local code = activation and tostring(activation.code or "") or ""
            if activationStatus == 403 and code == "hwid_mismatch" then
                fg100Notice("El código es para otro HWID. Registrá el HWID de este ejecutor.")
            elseif activationStatus == 403 and code == "unauthorized" then
                Environment.Young0xFG100AccessDenied = true
                if type(setFG100AccessVisual) == "function" then setFG100AccessVisual(false) end
                fg100PaidAccessNotice()
            elseif activationStatus == 429 then
                Environment.Young0xFG100RetryAt = os.clock() + 8
                fg100LoadingNotice()
            elseif activationStatus == 403 and code == "activation_expired" then
                clearUsedCode()
                fg100Notice("El código venció. Generá uno nuevo con /fg100 registrar.")
            elseif activationStatus == 403 and code == "invalid_activation" then
                clearUsedCode()
                fg100Notice("Ese código no coincide. Usá el último bloque de /fg100 registrar.")
            elseif activationStatus == 403 and code == "activation_required" then
                clearUsedCode()
                fg100Notice("Ese código ya no está disponible. Usá el último bloque de /fg100 registrar.")
            elseif fg100TransientStatus(activationStatus) then
                Environment.Young0xFG100RetryAt = os.clock() + 4
                fg100LoadingNotice()
            else
                fg100Notice("No se pudo confirmar la activación. Tu código sigue guardado")
            end
            return false
        end
        deviceToken = activation.deviceToken
        deviceNotSaved = not saveFG100DeviceToken(deviceToken, hwid)
        if deviceNotSaved then
            fg100Notice("FG100 quedó activado, pero este ejecutor no pudo guardar la credencial. Conservá este bloque para reintentarlo.")
        end
    end

    if not validFG100Session(session, sessionStatus) then
        session, sessionStatus = requestSession()
    end
    if not validFG100Session(session, sessionStatus) then
        return fg100SessionFailure(session, sessionStatus, hwid)
    end
    Environment.Young0xFG100AccessDenied = nil
    Environment.Young0xFG100RetryAt = nil
    if type(setFG100AccessVisual) == "function" then setFG100AccessVisual(true) end

    local payload, payloadStatus = fg100Post("/v1/fg100/payload", {
        sessionId = session.sessionId,
        sessionToken = session.sessionToken,
        context = fg100ExecutionContext(),
    })
    if fg100TransientStatus(payloadStatus) then
        fg100LoadingNotice()
        task.wait(FG100_RETRY_DELAYS[1])
        session, sessionStatus = requestSession()
        if validFG100Session(session, sessionStatus) then
            payload, payloadStatus = fg100Post("/v1/fg100/payload", {
                sessionId = session.sessionId,
                sessionToken = session.sessionToken,
                context = fg100ExecutionContext(),
            })
        else
            return fg100SessionFailure(session, sessionStatus, hwid)
        end
    end
    if payloadStatus ~= 200 or type(payload) ~= "string" or #payload < 64 then
        if fg100TransientStatus(payloadStatus) then
            Environment.Young0xFG100RetryAt = os.clock() + 4
            fg100LoadingNotice()
        else
            fg100Notice("No se pudo iniciar FG100. Probá nuevamente")
        end
        return false
    end

    Environment.Young0xFG100Auth = {
        userId = player.UserId,
        verified = true,
        expiresAt = os.time() + math.max(1, tonumber(session.expiresIn) or 30),
    }
    local chunk, compileError = loadstring(payload, "FG100")
    payload = nil
    if not chunk then
        Environment.Young0xFG100Auth = nil
        fg100Notice("Error al preparar FG100.")
        warn("[Young0x Hub] FG100 compile error: " .. tostring(compileError))
        return false
    end
    local runFinished, runSucceeded, runFailure = false, false, nil
    local okSpawn = pcall(task.spawn, function()
        runSucceeded, runFailure = pcall(chunk)
        chunk = nil
        runFinished = true
    end)
    if not okSpawn then
        Environment.Young0xFG100Auth = nil
        fg100Notice("FG100 no pudo iniciarse.")
        return false
    end

    local launchDeadline = os.clock() + 18
    local launchedGui = nil
    repeat
        launchedGui = playerGui:FindFirstChild("FG100Hub")
            or playerGui:FindFirstChild("Young0xFG100Hub")
        if launchedGui or (runFinished and not runSucceeded) then
            break
        end
        task.wait(0.05)
    until os.clock() >= launchDeadline

    launchedGui = launchedGui
        or playerGui:FindFirstChild("FG100Hub")
        or playerGui:FindFirstChild("Young0xFG100Hub")
    if not launchedGui then
        Environment.Young0xFG100Auth = nil
        if runFinished and not runSucceeded then
            warn("[Young0x Hub] FG100 runtime error: " .. tostring(runFailure))
        else
            warn("[Young0x Hub] FG100 launch confirmation timed out")
        end
        fg100LoadingNotice()
        return false
    end

    task.spawn(function()
        while not runFinished do
            task.wait(0.1)
        end
        if not runSucceeded then
            Environment.Young0xFG100Auth = nil
            warn("[Young0x Hub] FG100 runtime error after launch: " .. tostring(runFailure))
        end
    end)

    if deviceNotSaved then
        fg100Notice("FG100 activado. Tu ejecutor no pudo guardarlo en disco; la credencial se conserva durante esta sesión.")
    end
    clearUsedCode()
    return true
end

local FG100_AUTOSTART_ATTEMPTS = 3

local function runFG100Secure()
    local current = Environment.Young0xFG100Starting
    if type(current) == "table" and type(current.startedAt) == "number"
        and os.clock() - current.startedAt < 15 then
        fg100LoadingNotice()
        return false
    end

    local attempt = { startedAt = os.clock() }
    Environment.Young0xFG100Starting = attempt
    local ok, result = true, false
    for automaticAttempt = 1, FG100_AUTOSTART_ATTEMPTS do
        local retryAt = tonumber(Environment.Young0xFG100RetryAt) or 0
        if retryAt > os.clock() then
            fg100LoadingNotice()
            task.wait(math.max(0, retryAt - os.clock()))
        end

        local existing = playerGui:FindFirstChild("FG100Hub")
            or playerGui:FindFirstChild("Young0xFG100Hub")
        if existing then
            result = true
            break
        end

        Environment.Young0xFG100RetryAt = nil
        ok, result = pcall(runFG100SecureRequest)
        if not ok or result == true then
            break
        end

        retryAt = tonumber(Environment.Young0xFG100RetryAt) or 0
        if retryAt <= os.clock() or automaticAttempt >= FG100_AUTOSTART_ATTEMPTS then
            break
        end
    end
    if Environment.Young0xFG100Starting == attempt then
        Environment.Young0xFG100Starting = nil
    end
    if not ok then
        fg100LoadingNotice()
        warn("[Young0x Hub] FG100 start error: " .. tostring(result))
        return false
    end
    return result == true
end

local function runModule(file)
    if moduleInMaintenance(file) then

    if file == "000.lua" then
        return runFG100Secure()

    local ok, err = pcall(function()
        local url = RAW .. file
        local separator = string.find(url, "?", 1, true) and "&" or "?"
        url = url .. separator .. "v=" .. tostring(os.time())
        loadstring(game:HttpGet(url, true))()
    end)
    if not ok then
        warn("[Young0x Hub] " .. file .. ": " .. tostring(err))
    end
    return ok
end

    local function copyDiscord(fromShop)
        local clipboard = rawget(Environment, "setclipboard")
            or rawget(Environment, "toclipboard")
            or rawget(_G, "setclipboard")
            or rawget(_G, "toclipboard")
        if type(clipboard) == "function" then
            pcall(clipboard, DISCORD_URL)
        end
        if fromShop then
            showShopStatus("Link copiado")
        else
            showLoaderStatus("Link copiado", COLOR.Green, 2)
        end
    end

    local function animateButton(button, buttonScale, callback, hoverScale)
        track(button.MouseEnter:Connect(function()
            if not closing then
                playTween(buttonScale, 0.22, { Scale = hoverScale or 1.15 }, Enum.EasingStyle.Back)
            end
        end))
        track(button.MouseLeave:Connect(function()
            playTween(buttonScale, 0.2, { Scale = 1 }, Enum.EasingStyle.Quint)
        end))
        track(button.Activated:Connect(function()
            if closing or transitioning then
                return
            end
            playTween(buttonScale, 0.07, { Scale = 0.88 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
            task.delay(0.07, function()
                if buttonScale.Parent then
                    playTween(buttonScale, 0.2, { Scale = hoverScale or 1.15 }, Enum.EasingStyle.Back)
                end
            end)
            callback()
        end))
    end

        if state.module.maintenance then

        playTween(state.button, 0.26, {

        }, Enum.EasingStyle.Quint)

        playTween(state.icon, 0.26, {

            TextColor3 = hovering and COLOR.White or COLOR.Text,
        }, Enum.EasingStyle.Quint)
        playTween(state.dot, 0.28, {
            Size = hovering and UDim2.fromOffset(13, 13) or UDim2.fromOffset(10, 10),
        }, Enum.EasingStyle.Back)

    local function pulseModule(state)
        state.clickSerial += 1

        playTween(state.visualScale, 0.08, { Scale = 0.94 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)

            if serial == state.clickSerial and state.visualScale.Parent then
                playTween(state.visualScale, 0.24, {
                    Scale = state.hovered and 1.095 or 1,
                }, Enum.EasingStyle.Back)
            end
        end)
    end

    local function hasFG100LocalAccess()
        if playerGui:FindFirstChild("FG100Hub") or playerGui:FindFirstChild("Young0xFG100Hub") then
            return true
        end
        if Environment.Young0xFG100AccessDenied == true then
            return false
        end
        if type(Environment.Young0xFG100) == "table" then
            return true
        end
        local auth = Environment.Young0xFG100Auth
        if type(auth) == "table" and auth.verified == true
            and tostring(auth.userId) == tostring(player.UserId) then
            return true
        end
        if type(Environment.Young0xFG100ActivationCode) == "string"
            and #Environment.Young0xFG100ActivationCode >= 20
            and (Environment.Young0xFG100ActivationUserId == nil
                or tostring(Environment.Young0xFG100ActivationUserId) == tostring(player.UserId)) then
            return true
        end
        if type(gethwid) == "function" then
            local ok, hwid = pcall(gethwid)
            if ok and type(hwid) == "string" and #hwid >= 8 then
                local token = loadFG100DeviceToken(hwid)
                return validFG100DeviceToken(token)
            end
        end
        return false
    end

                    setFG100AccessVisual(true)

    local function exitAnimation(onComplete)
        if closing then
            return
        end
        closing = true
        dragging = false
        local exitTween = playTween(stage, 0.2, {
            GroupTransparency = 1,
            Position = UDim2.new(stage.Position.X.Scale, stage.Position.X.Offset, stage.Position.Y.Scale, stage.Position.Y.Offset + 18),
        }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        playTween(scale, 0.2, { Scale = scale.Scale * 0.93 }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        track(exitTween.Completed:Connect(function()
            destroyLoader()
            if onComplete then
                onComplete()
            end
        end))
    end

    closeOnly = function()
        if launching then
            return

        exitAnimation()
    end

    closeAndRun = function(file)
        exitAnimation(function()

    local function bindDrag(dragHeader)
        track(dragHeader.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragInput = input
                dragStart = input.Position
                dragOrigin = dragTarget
                track(input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end))
            end
        end))
    end
    bindDrag(header)
    bindDrag(titleButton)
    bindDrag(shopHeader)
    bindDrag(shopTitle)

    track(UserInputService.InputChanged:Connect(function(input)
        if dragging and not closing
            and (input == dragInput or input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            dragTarget = UDim2.new(
                dragOrigin.X.Scale,
                dragOrigin.X.Offset + delta.X,
                dragOrigin.Y.Scale,
                dragOrigin.Y.Offset + delta.Y
            )
        end
    end))

    track(RunService.RenderStepped:Connect(function(deltaTime)
        if not closing then
            local dragAlpha = 1 - math.exp(-10 * math.min(deltaTime, 0.05))
            stage.Position = stage.Position:Lerp(dragTarget, dragAlpha)
        end

        local pointer = UserInputService:GetMouseLocation()
        for _, state in ipairs(moduleStates) do
            local targetX = 0
            local targetY = 0
            if state.hovered and not state.module.maintenance and loaderPage.Visible then
                local absolutePosition = state.button.AbsolutePosition
                local absoluteSize = state.button.AbsoluteSize
                if absoluteSize.X > 0 and absoluteSize.Y > 0 then
                    local relativeX = math.clamp((pointer.X - absolutePosition.X) / absoluteSize.X - 0.5, -0.5, 0.5)
                    local relativeY = math.clamp((pointer.Y - absolutePosition.Y) / absoluteSize.Y - 0.5, -0.5, 0.5)
                    targetX = relativeX * 14
                    targetY = relativeY * 10
                end
            end
            local motionAlpha = 1 - math.exp(-13 * math.min(deltaTime, 0.05))
            state.offsetX += (targetX - state.offsetX) * motionAlpha
            state.offsetY += (targetY - state.offsetY) * motionAlpha
            state.visual.Position = UDim2.new(0.5, state.offsetX, 0, 2 + state.offsetY)
        end
    end))

    updateScale = function(animated)
        camera = workspace.CurrentCamera or camera
        if not camera then
            return
        end
        local viewport = camera.ViewportSize
        local mobile = clientUsesMobileLayout(viewport)
        for _, state in ipairs(moduleStates) do
            local iconSize = mobile
                and (state.module.mobileIconSize or state.module.iconSize or 84)
                or (state.module.iconSize or 84)
            local iconY = mobile
                and (state.module.mobileIconY or state.module.iconY or 57)
                or (state.module.iconY or 57)
            state.icon.Size = UDim2.fromOffset(iconSize, iconSize)
            state.icon.Position = UDim2.new(0.5, 0, 0, iconY)
        end
        local compactDesktop = not mobile
            and (viewport.X <= 1100 or viewport.Y <= 650)
        local horizontalPadding
        local verticalPadding
        local maximumScale
        local minimumScale

        if mobile then
            horizontalPadding = math.max(52, viewport.X * 0.18)
            verticalPadding = math.max(92, viewport.Y * 0.32)
            maximumScale = MOBILE_MAX_SCALE
            minimumScale = 0.22
        elseif compactDesktop then
            horizontalPadding = 28
            verticalPadding = 40
            maximumScale = COMPACT_DESKTOP_MAX_SCALE
            minimumScale = 0.30
        else
            horizontalPadding = 34
            verticalPadding = 46
            maximumScale = DESKTOP_MAX_SCALE
            minimumScale = 0.30
        end

        local availableWidth = math.max(viewport.X - horizontalPadding, 1)
        local availableHeight = math.max(viewport.Y - verticalPadding, 1)
        local fittedScale = math.min(
            availableWidth / currentWidth,
            availableHeight / currentHeight,
            maximumScale
        )
        local targetScale = math.max(fittedScale, minimumScale)
        if animated then
            playTween(scale, 0.3, { Scale = targetScale }, Enum.EasingStyle.Quint)
        else
            scale.Scale = targetScale
        end
    end

        if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
            object.Font = Enum.Font.Garamond
        end
    end

        applyTypography(object)
    end

        track(camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale))
    end
    track(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
        camera = workspace.CurrentCamera
        updateScale()
    end))

    playTween(scale, 0.36, { Scale = finalScale }, Enum.EasingStyle.Back)

                    if not closing and not launching and state.button.Parent then
                        state.button:Activate()
                    end
                end)
                break
            end
        end
    end

    return
