local library = library -- (Veya kütüphanenin tanımlandığı değişken adı)

local Window = library:AddWindow("MAMI HUB V2", {
    main_color = Color3.fromRGB(41, 74, 122),
    min_size = Vector2.new(450, 350),
    toggle_key = Enum.KeyCode.RightShift,
    can_resize = true,
})

local MainTab = Window:AddTab("Ana Sayfa")

MainTab:AddLabel("MAMI HUB V2'ye Hoş Geldiniz!")

MainTab:AddButton("Örnek Buton", function()
    print("Butona tıklandı!")
end)

MainTab:AddSwitch("Özellik Aç/Kapat", function(state)
    print("Durum:", state)
end)
