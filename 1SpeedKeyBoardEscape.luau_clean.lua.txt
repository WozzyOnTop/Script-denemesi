  WinPos = Vector3.new(-6809.3223, 531.2539, 1468.8073),
  WinSpeed = 300,

Color3.fromHex("#0400ff")

local function f1(p1)
  local g = p1.G
  local b = p1.B
  local r = p1.R
  local v4 = (r + g + b) / 3
  return math.abs(r - v4) + math.abs(g - v4) + math.abs(b - v4) < 0.08 or v4 < 0.05 or v4 > 0.95
end

local connect = nil
local bodyVelocity, bodyGyro

local function f2()
  v3.Fly = false

  if connect then
    connect:Disconnect()
    connect = nil
  end

  if bodyVelocity then
    bodyVelocity:Destroy()
    bodyVelocity = nil
  end

  if bodyGyro then
    bodyGyro:Destroy()
    bodyGyro = nil
  end

  local humanoid = localPlayer.Character
    and localPlayer.Character:FindFirstChildOfClass("Humanoid")

  if humanoid then
    humanoid.PlatformStand = false
    humanoid:ChangeState(Enum.HumanoidStateType.Running)
  end
end

local function f3()
  local character = localPlayer.Character
  local humanoidRootPart = character and character:FindFirstChild("HumanoidRootPart")

  local humanoid2 = character
  humanoid2 = character and character:FindFirstChildOfClass("Humanoid")

  local v5 = not humanoidRootPart
  local v6 = humanoid2

  if v5 or not v6 then
    return
  end

  if bodyVelocity then
    bodyVelocity:Destroy()
  end

  if bodyGyro then
    bodyGyro:Destroy()
  end

  if connect then
    connect:Disconnect()
  end

  v6.PlatformStand = true

  bodyGyro = Instance.new("BodyGyro")
  bodyGyro.MaxTorque = Vector3.new(9000000000, 9000000000, 9000000000)
  bodyGyro.P = 50000
  bodyGyro.CFrame = humanoidRootPart.CFrame
  bodyGyro.Parent = humanoidRootPart

  bodyVelocity = Instance.new("BodyVelocity")
  bodyVelocity.MaxForce = Vector3.new(9000000000, 9000000000, 9000000000)
  bodyVelocity.Velocity = Vector3.zero
  bodyVelocity.Parent = humanoidRootPart

  connect = runService.RenderStepped:Connect(function()
    if not v3.Fly then
      return
    else
      local cframe = currentCamera.CFrame
      local flySpeed = v3.FlySpeed
      local zero = Vector3.zero

      if userInputService:IsKeyDown(Enum.KeyCode.W) then
        zero = zero + cframe.LookVector * flySpeed
      end

      if userInputService:IsKeyDown(Enum.KeyCode.S) then
        zero = zero - cframe.LookVector * flySpeed
      end

      if userInputService:IsKeyDown(Enum.KeyCode.A) then
        zero = zero - cframe.RightVector * flySpeed
      end

      if userInputService:IsKeyDown(Enum.KeyCode.D) then
        zero = zero + cframe.RightVector * flySpeed
      end

      if userInputService:IsKeyDown(Enum.KeyCode.Space) then
        zero = zero + Vector3.yAxis * flySpeed
      end

      if userInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
        zero = zero - Vector3.yAxis * flySpeed
      end

      bodyVelocity.Velocity = zero
      bodyGyro.CFrame = cframe
      return
    end
  end)
end

local color

task.delay(1, function()
  local v7 = {}

  local function f4(p2)
    for key, value in pairs(p2:GetDescendants()) do
      local v8 = value

      pcall(function()
        local v9 = {}

        if v8:IsA("Frame") or v8:IsA("TextButton") or v8:IsA("ImageButton") then
          table.insert(v9, v8.BackgroundColor3)
        end

        if v8:IsA("ImageLabel") or v8:IsA("ImageButton") then
          table.insert(v9, v8.ImageColor3)
        end

        if v8:IsA("UIStroke") then
          table.insert(v9, v8.Color)
        end

        for key2, value2 in pairs(v9) do
          if not f1(value2) then
            local v10 = math.floor(value2.R * 20) .. "_" .. math.floor(value2.G * 20) .. "_"
              .. math.floor(value2.B * 20)

            v7[v10] = v7[v10] or { count = 0, color = value2 }

            local v11 = v7[v10]
            v11.count = v11.count + 1
          end
        end
      end)
    end
  end

  f4(coreGui)

  pcall(function() f4(players.LocalPlayer.PlayerGui) end)
  local count = 0
  local v12

  for key3, value3 in pairs(v7) do
    if value3.count > count then
      count = value3.count
      v12 = key3
    end
  end

  if v12 and v7[v12] then
    color = v7[v12].color
  end
end)

autoFarmTab:Toggle({
  Title = "Auto Win (Smooth Travel)",
  Value = false,
  Callback = function(value5)
    v3.AutoWin = value5

    if value5 then
      task.spawn(function()
        while v3.AutoWin do
          local character3 = localPlayer.Character
          local humanoidRootPart2 = character3 and character3:FindFirstChild("HumanoidRootPart")

          if humanoidRootPart2 and v3.WinPos then
            if (humanoidRootPart2.Position - v3.WinPos).Magnitude > 5 then
              v3.Fly = true
              f3()
              local noclip = v3.Noclip
              v3.Noclip = true
              horizonLibary:Notify({ Title = "Auto Win", Content = "Rising up...", Duration = 2 })
              local cframe2 = humanoidRootPart2.CFrame
              local v13 = cframe2 * CFrame.new(0, v3.WinHeight, 0)

              for i = 0, 1, 0.05 do
                if not v3.AutoWin or not humanoidRootPart2.Parent then
                  break
                end

                humanoidRootPart2.CFrame = cframe2:Lerp(v13, i)
                task.wait(0.02)
              end

              if not v3.AutoWin or not humanoidRootPart2.Parent then
                v3.Noclip = noclip
                return
              else
                task.wait(0.2)

                horizonLibary:Notify({
                  Title = "Auto Win",
                  Content = "Traveling...",
                  Duration = 2,
                })

                local cframe3 = humanoidRootPart2.CFrame

                local vector = Vector3.new(
                  v3.WinPos.X, humanoidRootPart2.Position.Y, v3.WinPos.Z
                )

                local cframe4 = CFrame.new(vector)

                local v14 = math.max(1, math.floor((humanoidRootPart2.Position - vector).Magnitude
                  / (v3.WinSpeed / 30)))

                for j = 1, v14 do
                  if not v3.AutoWin or not humanoidRootPart2.Parent then
                    break
                  end

                  humanoidRootPart2.CFrame = cframe3:Lerp(cframe4, j / v14)
                  task.wait(0.03)
                end

                if not v3.AutoWin or not humanoidRootPart2.Parent then
                  v3.Noclip = noclip
                  return
                end

                task.wait(0.2)

                horizonLibary:Notify({
                  Title = "Auto Win",
                  Content = "Descending...",
                  Duration = 2,
                })

                local cframe5 = humanoidRootPart2.CFrame
                local cframe6 = CFrame.new(v3.WinPos)

                for k = 0, 1, 0.01 do
                  if not v3.AutoWin or not humanoidRootPart2.Parent then
                    break
                  end

                  humanoidRootPart2.CFrame = cframe5:Lerp(cframe6, k)
                  task.wait(0.01)
                end

                if humanoidRootPart2.Parent then
                  humanoidRootPart2.CFrame = cframe6
                end

                v3.Noclip = noclip
                f2()

                horizonLibary:Notify({
                  Title = "Auto Win",
                  Content = "Arrived! Resetting in 3s...",
                  Duration = 3,
                })

                task.wait(3)

                local humanoid4 = localPlayer.Character
                  and localPlayer.Character:FindFirstChildOfClass("Humanoid")

                if humanoid4 then
                  humanoid4.Health = 0
                end

                horizonLibary:Notify({
                  Title = "Auto Win",
                  Content = "Next loop in 5s...",
                  Duration = 5,
                })

                task.wait(5)
              end
            end
          end

          task.wait(0.1)
        end
      end)
    end
  end,
})
