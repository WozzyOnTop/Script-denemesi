-- WindUI Kütüphanesini Yükle
local WindUI = loadstring(game:HttpGet("https://github.com/WozzyOnTop/Script-denemesi/raw/refs/heads/main/wind%20ui.txt"))()

-- Yeni bir pencere (Window) oluştur
local Window = WindUI:CreateWindow({
    Title = "Arayüz Başlığı",
    Icon = "rbxassetid://75920162824531", -- İsteğe bağlı ikon
    Author = "Yazar Adı",
    Folder = "MyScriptConfig",
    Size = UDim2.fromOffset(580, 460),
    Transparent = true,
    Theme = "Dark", -- Dark, Light, Rose, Plant, Red, Indigo, Sky, Violet, Amber vb.
    SideBarWidth = 200,
    HasOutline = true,
})

-- Bir Sekme (Tab) Ekle
local MainTab = Window:Tab({
    Title = "Ana Sayfa",
    Icon = "home",
})

-- Sekme İçine Bir Bölüm (Section) Ekle
local Section = MainTab:Section({
    Title = "Genel Ayarlar",
})

-- Buton Ekleme
Section:Button({
    Title = "Buton",
    Desc = "Tıklandığında bir işlem çalıştırır",
    Callback = function()
        print("Butona tıklandı!")
    end
})

-- Toggle (Açma/Kapama) Ekleme
Section:Toggle({
    Title = "Özellik Aç/Kapat",
    Value = false,
    Callback = function(state)
        print("Durum:", state)
    end
})

-- Bildirim GöndermeÖrneği
WindUI:Notify({
    Title = "Başarılı",
    Content = "WindUI arayüzü başarıyla yüklendi!",
    Duration = 5,
})
