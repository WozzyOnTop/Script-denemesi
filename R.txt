local function getRole(plr)
	if not plr then return "Unknown" end
	-- attributes
	local ok, attr = pcall(function()
		return plr:GetAttribute("Role") or plr:GetAttribute("role") or plr:GetAttribute("CurrentRole")
	end)
	if ok and attr and tostring(attr) ~= "" and tostring(attr) ~= "None" then
		return tostring(attr)
	end
	-- common MM2 values / folders
	pcall(function()
		local ls = plr:FindFirstChild("PlayerData") or plr:FindFirstChild("Data") or plr:FindFirstChild("leaderstats")
		if ls then
			local r = ls:FindFirstChild("Role") or ls:FindFirstChild("role")
			if r then
				if r:IsA("StringValue") or r:IsA("StringValue") then attr = r.Value
				elseif r:IsA("ValueBase") then attr = r.Value end
			end
		end
	end)
	if attr and tostring(attr) ~= "" then return tostring(attr) end
	pcall(function()
		local rs = game:GetService("ReplicatedStorage")
		for _, name in ipairs({"Roles", "Role", "GameData", "RoundData", "Remotes"}) do
			local folder = rs:FindFirstChild(name, true)
			if folder then
				local v = folder:FindFirstChild(plr.Name) or folder:FindFirstChild(tostring(plr.UserId))
				if v then
					if v:IsA("StringValue") then attr = v.Value
					elseif v:GetAttribute("Role") then attr = v:GetAttribute("Role")
					elseif v:IsA("ValueBase") then attr = v.Value end
				end
			end
		end
	end)
	if attr and tostring(attr) ~= "" then return tostring(attr) end
	local function hasTool(name)
		name = name:lower()
		local function scan(container)
			if not container then return false end
			for _, t in ipairs(container:GetChildren()) do
				if t:IsA("Tool") then
					local tn = t.Name:lower()
					if tn:find(name, 1, true) or tn == name then return true end
				end
			end
			return false
		end
		return scan(plr.Character) or scan(plr:FindFirstChildOfClass("Backpack"))
	end
	if hasTool("knife") or hasTool("gunknife") then return "Murderer" end
	if hasTool("gun") or hasTool("revolver") or hasTool("pistol") then return "Sheriff" end
	return "Innocent"
end

local function isMurderer(plr)
	return tostring(getRole(plr)):lower():find("murder") ~= nil
end
local function isSheriff(plr)
	local r = tostring(getRole(plr)):lower()
	return r:find("sheriff") ~= nil or r:find("hero") ~= nil
end

local function equipTool(sub)
	sub = sub:lower()
	local char = LP.Character
	local pack = LP:FindFirstChildOfClass("Backpack")
	local function try(container)
		if not container then return nil end
		for _, t in ipairs(container:GetChildren()) do
			if t:IsA("Tool") and t.Name:lower():find(sub, 1, true) then
				pcall(function()
					local h = getHum()
					if h then h:EquipTool(t) end
				end)
				return t
			end
		end
		return nil
	end
	return try(char) or try(pack)
end

local function activateTool()
	local char = LP.Character
	if not char then return end
	local tool = char:FindFirstChildOfClass("Tool")
	if not tool then return end
	pcall(function() tool:Activate() end)
	-- try common tool remotes
	pcall(function()
		for _, r in ipairs(tool:GetDescendants()) do
			if r:IsA("RemoteEvent") then
				pcall(function() r:FireServer() end)
				pcall(function() r:FireServer("Slash") end)
				pcall(function() r:FireServer("Shoot") end)
			end
		end
	end)
	-- mouse click simulation for gun
	pcall(function()
		local vim = game:GetService("VirtualInputManager")
		if vim then
			vim:SendMouseButtonEvent(0, 0, 0, true, game, 0)
			task.wait()
			vim:SendMouseButtonEvent(0, 0, 0, false, game, 0)
		end
	end)
end

-- anti-hit (lightweight, safe)
local AH = { fake = nil, step = nil, original = nil }
local function stopAntiHit()
	S.antiHit = false
	if AH.step then disc(AH.step) AH.step = nil end
	if AH.fake then pcall(function() AH.fake:Destroy() end) AH.fake = nil end
	local hrp = getHRP()
	if hrp and AH.original then pcall(function() hrp.CFrame = AH.original end) end
end
local function startAntiHit()
	S.antiHit = true
	local hrp = getHRP()
	if not hrp then return end
	AH.original = hrp.CFrame
	if AH.step then disc(AH.step) end
	AH.step = bind(RunService.Heartbeat:Connect(function()
		if not S.antiHit then return end
		local root = getHRP()
		if not root then return end
		-- soft desync offset
		pcall(function()
			root.CFrame = root.CFrame + Vector3.new(0, 0.01, 0)
			root.CFrame = root.CFrame - Vector3.new(0, 0.01, 0)
		end)
	end))
end

local aimbotFOVCircle = nil
local function getAimPart(char, partName)
	if not char then return nil end
	partName = partName or "Head"
	local p = char:FindFirstChild(partName)
	if p and p:IsA("BasePart") then return p end
	if partName == "Torso" then
		p = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
		if p then return p end
	end
	return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local function isOnScreen(worldPos, fov)
	local cam = Workspace.CurrentCamera
	if not cam then return false, 999 end
	local sp, onScreen = cam:WorldToViewportPoint(worldPos)
	if not onScreen or sp.Z < 0 then return false, 999 end
	local center = cam.ViewportSize / 2
	local dist = (Vector2.new(sp.X, sp.Y) - Vector2.new(center.X, center.Y)).Magnitude
	local maxR = (fov / 90) * (math.min(center.X, center.Y) * 0.6)
	return dist <= maxR, dist
end

local function pickAimTarget()
	local cam = Workspace.CurrentCamera
	local my = getHRP()
	if not cam or not my then return nil end
	local best, bestDist = nil, 1e9
	local mode = S.aimbotTeam or "murder"
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character then
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health > 0 then
				local okTarget = false
				if mode == "murder" then
					okTarget = isMurderer(plr)
				elseif mode == "nearest" then
					okTarget = true
				elseif mode == "all" then
					okTarget = not isSheriff(plr) -- don't lock teammates-ish
				else
					okTarget = isMurderer(plr)
				end
				if okTarget then
					local part = getAimPart(plr.Character, S.aimbotPart)
					if part then
						local pos = part.Position
						if S.aimbotPred then
							local vel = Vector3.zero
							pcall(function() vel = part.AssemblyLinearVelocity end)
							pos = pos + vel * (S.aimbotPredAmt or 0.12)
						end
						local inFov, screenDist = isOnScreen(pos, S.aimbotFOV or 120)
						if inFov and screenDist < bestDist then
							-- optional visibility
							local visible = true
							if S.aimbotVisible then
								local origin = cam.CFrame.Position
								local dir = (pos - origin)
								local rayParams = RaycastParams.new()
								rayParams.FilterType = Enum.RaycastFilterType.Exclude
								rayParams.FilterDescendantsInstances = {LP.Character}
								local hit = Workspace:Raycast(origin, dir, rayParams)
								if hit and hit.Instance and not hit.Instance:IsDescendantOf(plr.Character) then
									visible = false
								end
							end
							if visible then
								best = {plr = plr, part = part, pos = pos}
								bestDist = screenDist
							end
						end
					end
				end
			end
		end
	end
	return best
end

local function applyAimbot()
	if not S.aimbot then return end
	pcall(function()
		local cam = Workspace.CurrentCamera
		if cam and cam.CameraType ~= Enum.CameraType.Custom and cam.CameraType ~= Enum.CameraType.Scriptable then
			cam.CameraType = Enum.CameraType.Custom
		end
	end)
	-- optional key hold (right mouse / touch always ok for mobile)
	if S.aimbotKey then
		local holding = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
			or UserInputService:IsKeyDown(Enum.KeyCode.E)
		if not holding and not UserInputService.TouchEnabled then return end
	end
	local cam = Workspace.CurrentCamera
	if not cam then return end
	local target = S.aimbotTarget
	if (not target) or (not S.aimbotSticky) or (not target.plr) or (not target.plr.Character) then
		target = pickAimTarget()
		S.aimbotTarget = target
	else
		-- refresh pos on sticky
		local part = getAimPart(target.plr.Character, S.aimbotPart)
		if part then
			local pos = part.Position
			if S.aimbotPred then
				local vel = Vector3.zero
				pcall(function() vel = part.AssemblyLinearVelocity end)
				pos = pos + vel * (S.aimbotPredAmt or 0.12)
			end
			target.part = part
			target.pos = pos
		else
			target = pickAimTarget()
			S.aimbotTarget = target
		end
	end
	if not target or not target.pos then return end
	local smooth = tonumber(S.aimbotSmooth) or 0.35
	smooth = math.clamp(smooth, 0.05, 1)
	local goal = CFrame.new(cam.CFrame.Position, target.pos)
	cam.CFrame = cam.CFrame:Lerp(goal, smooth)
	if S.aimbotAutoShoot then
		equipTool("gun")
		activateTool()
	end
end

local function updateFOVCircle()
	if not S.aimbotDrawFOV then
		if aimbotFOVCircle then pcall(function() aimbotFOVCircle:Destroy() end) aimbotFOVCircle = nil end
		return
	end
	local cam = Workspace.CurrentCamera
	if not cam then return end
	if not aimbotFOVCircle then
		local parent = CoreGui
		pcall(function()
			if type(gethui) == "function" then parent = gethui() or parent end
		end)
		local sg = Instance.new("ScreenGui")
		sg.Name = "EH_AimbotFOV"
		sg.ResetOnSpawn = false
		sg.IgnoreGuiInset = true
		sg.Parent = parent
		local fr = Instance.new("Frame")
		fr.AnchorPoint = Vector2.new(0.5, 0.5)
		fr.Position = UDim2.fromScale(0.5, 0.5)
		fr.BackgroundTransparency = 1
		fr.Parent = sg
		local stroke = Instance.new("UIStroke")
		stroke.Color = Color3.fromRGB(90, 140, 255)
		stroke.Thickness = 1.5
		stroke.Parent = fr
		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(1, 0)
		corner.Parent = fr
		aimbotFOVCircle = sg
		aimbotFOVCircle._frame = fr
	end
	local center = cam.ViewportSize / 2
	local radius = (S.aimbotFOV / 90) * (math.min(center.X, center.Y) * 0.6) * 2
	local fr = aimbotFOVCircle._frame
	if fr then
		fr.Size = UDim2.fromOffset(radius, radius)
	end
end

			if f["ab_pr_" .. i] then
				S.aimbotPred = true
				S.aimbotPredAmt = i * 0.01
			end
		end
		pcall(updateFOVCircle)

		-- kill aura ranges from flags
		if S.auraRange then S.killRange = S.auraRange end

		-- core kill aura
		if S.killAura and my then
			equipTool("knife")
			local range = S.killRange or S.auraRange or 12
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					local h = plr.Character:FindFirstChildOfClass("Humanoid")
					if r and h and h.Health > 0 and (r.Position - my.Position).Magnitude <= range then
						activateTool()
					end
				end
			end
		end

		if S.autoKnife or f["knife_swing"] then
			equipTool("knife")
			activateTool()
		end

		-- knife swing rates
		for ms = 1, 20 do
			if f["swing_" .. ms] then
				equipTool("knife")
				activateTool()
			end
		end

		-- auto shoot murderer (core + flags)
		if (S.autoShoot or f["s_shoot"] or f["gun_sos"]) and my then
			local range = S.shootRange or 90
			for r = 30, 200, 5 do
				if f["s_range_" .. r] or f["gun_r_" .. r] then range = r end
			end
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and isMurderer(plr) and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r and (r.Position - my.Position).Magnitude <= range then
						equipTool("gun")
						activateTool()
						if f["s_face"] then
							pcall(function() my.CFrame = CFrame.new(my.Position, r.Position) end)
						end
					end
				end
			end
		end

		-- chase
		if f["chase_any"] or f["chase_soft"] or f["chase_hard"] or f["chase_sheriff"] or f["chase_hero"] then
			if my then
				local maxD = S.chaseDist or 90
				local best, bd = nil, maxD
				for _, plr in ipairs(Players:GetPlayers()) do
					if plr ~= LP and plr.Character then
						local skip = false
						if f["chase_sheriff"] and not isSheriff(plr) then skip = true end
						if f["chase_hero"] and not tostring(getRole(plr)):lower():find("hero") then skip = true end
						if not skip then
							local r = plr.Character:FindFirstChild("HumanoidRootPart")
							local h = plr.Character:FindFirstChildOfClass("Humanoid")
							if r and h and h.Health > 0 then
								local d = (r.Position - my.Position).Magnitude
								if d < bd then best, bd = r, d end
							end
						end
					end
				end
				if best then
					local alpha = f["chase_hard"] and 0.4 or 0.22
					pcall(function() my.CFrame = my.CFrame:Lerp(CFrame.new(best.Position + Vector3.new(0, 0, 3)), alpha) end)
					if f["face_target"] then pcall(function() my.CFrame = CFrame.new(my.Position, best.Position) end) end
					equipTool("knife")
					activateTool()
				end
			end
		end

		if f["backstab"] and my then
			local best, bd = nil, 18
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r then
						local d = (r.Position - my.Position).Magnitude
						if d < bd then best, bd = r, d end
					end
				end
			end
			if best then
				pcall(function() my.CFrame = best.CFrame * CFrame.new(0, 0, 3.2) end)
				equipTool("knife")
				activateTool()
			end
		end

		if f["tp_stab"] and my then
			local best, bd = nil, 60
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r then
						local d = (r.Position - my.Position).Magnitude
						if d < bd then best, bd = r, d end
					end
				end
			end
			if best then
				pcall(function() my.CFrame = best.CFrame + Vector3.new(0, 2, 0) end)
				equipTool("knife")
				activateTool()
			end
		end

		if f["orbit"] and my then
			local rad = tonumber(f["orbit_rad"]) or 8
			local best, bd = nil, 50
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r then
						local d = (r.Position - my.Position).Magnitude
						if d < bd then best, bd = r, d end
					end
				end
			end
			if best then
				S._ang = (S._ang or 0) + 0.18
				local pos = best.Position + Vector3.new(math.cos(S._ang) * rad, 0, math.sin(S._ang) * rad)
				pcall(function() my.CFrame = CFrame.new(pos, best.Position) end)
				equipTool("knife")
				activateTool()
			end
		end

		-- sheriff track face
		if f["s_track"] or f["s_face"] then
			if my then
				for _, plr in ipairs(Players:GetPlayers()) do
					if plr ~= LP and isMurderer(plr) and plr.Character then
						local r = plr.Character:FindFirstChild("HumanoidRootPart")
						if r then
							pcall(function() my.CFrame = CFrame.new(my.Position, r.Position) end)
						end
					end
				end
			end
		end

		if S.autoGun or f["s_keep"] or f["gun_keep"] then equipTool("gun") end
		if f["knife_keep"] then equipTool("knife") end

		if f["s_grab"] or f["s_camp"] or f["gun_grab"] or f["gun_camp"] then
			if my then
				for _, d in ipairs(Workspace:GetDescendants()) do
					if d:IsA("Tool") then
						local n = d.Name:lower()
						if (n:find("gun") or n:find("revolver") or n:find("pistol")) and d.Parent ~= LP.Character and d.Parent ~= LP:FindFirstChildOfClass("Backpack") then
							local p = d:FindFirstChildWhichIsA("BasePart")
							if p then pcall(function() my.CFrame = p.CFrame + Vector3.new(0, 2.5, 0) end) end
						end
					end
				end
			end
		end

		-- innocent flee
		if f["i_run"] and my then
			local dist = S.fleeDist or 40
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and isMurderer(plr) and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r and (r.Position - my.Position).Magnitude < dist then
						local away = my.Position - r.Position
						if away.Magnitude > 0.1 then
							pcall(function() my.CFrame = CFrame.new(my.Position + away.Unit * 14) end)
						end
						if hum then hum.WalkSpeed = math.max(hum.WalkSpeed, 26) end
					end
				end
			end
		end

		if f["i_follow_s"] and my then
			for _, plr in ipairs(Players:GetPlayers()) do
				if isSheriff(plr) then
					local r = getRoot(plr)
					if r then pcall(function() my.CFrame = r.CFrame * CFrame.new(0, 0, 5) end) end
				end
			end
		end
		if f["i_follow_h"] and my then
			for _, plr in ipairs(Players:GetPlayers()) do
				if tostring(getRole(plr)):lower():find("hero") then
					local r = getRoot(plr)
					if r then pcall(function() my.CFrame = r.CFrame * CFrame.new(0, 0, 5) end) end
				end
			end
		end

		if f["i_alert"] or f["n_murder_near"] then
			local ad = S.alertDist or 40
			for _, plr in ipairs(Players:GetPlayers()) do
				if isMurderer(plr) then
					local r = getRoot(plr)
					if r and my and (r.Position - my.Position).Magnitude < ad then
						if not S._alerted then
							S._alerted = true
							notify("DANGER", "Murderer near: " .. plr.Name)
						end
					else
						S._alerted = false
					end
				end
			end
		end

		-- coins
		if (S.coinFarm or f["coin_tp"] or f["coin_farm2"]) and my then
			local best, bd = nil, 1e9
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") then
					local n = d.Name:lower()
					if n:find("coin") or n:find("gold") then
						local dist = (d.Position - my.Position).Magnitude
						if dist < bd then best, bd = d, dist end
					end
				end
			end
			if best then pcall(function() my.CFrame = best.CFrame + Vector3.new(0, 2, 0) end) end
		end
		if f["coin_magnet"] and my then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("coin") then
					pcall(function() d.CFrame = my.CFrame end)
				end
			end
		end

		-- movement from flags
		if hum then
			local appliedSpeed = false
			for spd = 16, 80 do
				if f["spd_" .. spd] then
					hum.WalkSpeed = spd
					S.speed = spd
					appliedSpeed = true
				end
			end
			if S.speedOn and not appliedSpeed then
				hum.WalkSpeed = S.speed
			end
			local appliedJump = false
			for jp = 50, 200, 2 do
				if f["jp_" .. jp] then
					hum.JumpPower = jp
					pcall(function() hum.JumpHeight = jp / 7.2 end)
					appliedJump = true
				end
			end
			if S.jumpOn and not appliedJump then
				hum.JumpPower = S.jump
			end
			for h = 0, 10 do
				if f["hip_" .. h] then pcall(function() hum.HipHeight = h end) end
			end
		end

		-- fly speeds from flags
		for fs = 20, 150, 5 do
			if f["fly_" .. fs] then
				S.flySpeed = fs
				if not S.fly then setFly(true) end
			end
		end
		if f["fly_on"] and not S.fly then setFly(true) end

		if f["noclip_full"] and not S.noclip then setNoclip(true) end
		if f["spin"] and my then
			my.CFrame = my.CFrame * CFrame.Angles(0, math.rad(14), 0)
		end
		if (f["bhop"] or f["infjump"]) and hum and hum.FloorMaterial ~= Enum.Material.Air then
			hum.Jump = true
		end

		-- protect
		if f["prot_stun"] or f["prot_fall"] or S.antiStun then
			if hum then
				pcall(function()
					hum.PlatformStand = false
					hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
					hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
					if hum:GetState() == Enum.HumanoidStateType.Physics then
						hum:ChangeState(Enum.HumanoidStateType.Running)
					end
				end)
			end
			if my and my.Anchored then pcall(function() my.Anchored = false end) end
		end
		if f["prot_heal"] and hum then
			pcall(function() hum.Health = hum.MaxHealth end)
		end
		if f["prot_afk"] or f["misc_afk"] or S.antiAFK then
			pcall(function()
				VirtualUser:CaptureController()
				VirtualUser:ClickButton2(Vector2.new())
			end)
		end

		-- world / gfx
		if f["world_door"] or f["world_door2"] or f["noclip_door"] then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("door") then
					d.CanCollide = false
				end
			end
		end
		if f["gfx_fb"] or f["world_fb"] or S.fullbright then
			pcall(function()
				Lighting.Brightness = 2
				Lighting.ClockTime = 14
				Lighting.FogEnd = 1e6
			end)
		end
		if f["gfx_nf"] or f["world_nf"] or S.noFog then
			pcall(function() Lighting.FogEnd = 1e6 end)
		end
		if f["gfx_ns"] or f["world_ns"] then
			pcall(function() Lighting.GlobalShadows = false end)
		end
		for b = 1, 20 do
			if f["bright_" .. b] then pcall(function() Lighting.Brightness = b end) end
		end
		for c = 0, 24 do
			if f["clock_" .. c] then pcall(function() Lighting.ClockTime = c end) end
		end
		for fov = 50, 120 do
			if f["fov_" .. fov] then
				pcall(function() Workspace.CurrentCamera.FieldOfView = fov end)
			end
		end

		-- knife hitbox
		if f["knife_sz_on"] and S.knifeSize then
			local size = S.knifeSize
			local char = LP.Character
			local tool = char and char:FindFirstChildOfClass("Tool")
			if tool and tool.Name:lower():find("knife") then
				for _, p in ipairs(tool:GetDescendants()) do
					if p:IsA("BasePart") then
						p.Size = Vector3.new(size, size, size)
						p.Massless = true
						p.Transparency = 0.65
						p.CanCollide = false
					end
				end
			end
		end

		-- player hitbox expand
		if f["hitbox_on"] and S.hitboxSize then
			local i = S.hitboxSize
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
					if hrp then
						pcall(function()
							hrp.Size = Vector3.new(i, i, i)
							hrp.Transparency = 0.7
							hrp.CanCollide = false
							hrp.Massless = true
						end)
					end
				end
			end
		end

		-- ESP master flags
		if f["esp_master"] or f["esp_m"] or f["esp_s"] or f["esp_all"] then
			if f["esp_m"] then S.espMurder = true end
			if f["esp_s"] then S.espSheriff = true end
			if f["esp_i"] then S.espInnocent = true end
			if not S.esp then setESP(true) end
		end

		end) -- pcall loop body
		if not okLoop then
			warn("[Legacy Hub] loop:", errLoop)
		end
	end
end)

local function makePage(name)
	local sc = Instance.new("ScrollingFrame")
	sc.Name = name
	sc.Size = UDim2.fromScale(1,1)
	sc.BackgroundTransparency = 1
	sc.BorderSizePixel = 0
	sc.ScrollBarThickness = 3
	sc.CanvasSize = UDim2.new(0,0,0,0)
	sc.Visible = false
	sc.Parent = Content
	local lay = Instance.new("UIListLayout", sc)
	lay.Padding = UDim.new(0, 6)
	lay.SortOrder = Enum.SortOrder.LayoutOrder
	local pad = Instance.new("UIPadding", sc)
	pad.PaddingTop = UDim.new(0, 8)
	pad.PaddingLeft = UDim.new(0, 8)
	pad.PaddingRight = UDim.new(0, 8)
	lay:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		sc.CanvasSize = UDim2.new(0,0,0, lay.AbsoluteContentSize.Y + 16)
	end)
	pages[name] = sc
	return sc
end

local function selectTab(name)
	currentTab = name
	for n, p in pairs(pages) do p.Visible = (n == name) end
	for n, b in pairs(tabButtons) do
		b.BackgroundColor3 = (n == name) and COLORS.tabActive or COLORS.tab
		b.TextColor3 = (n == name) and COLORS.accent or COLORS.muted
	end
end

local function addTab(name, order)
	local b = Instance.new("TextButton")
	b.Size = UDim2.new(1,0,0,28)

	makePage(name)

end

local function numberBox(page, label, order, default, minV, maxV, onApply)
	local hold = Instance.new("Frame")
	hold.Size = UDim2.new(1, -8, 0, 56)
	hold.BackgroundColor3 = Color3.fromRGB(18, 20, 36)
	hold.LayoutOrder = order
	hold.Parent = page
	corner(hold, 8)
	local lab = Instance.new("TextLabel", hold)
	lab.Size = UDim2.new(1, -16, 0, 18)
	lab.Position = UDim2.fromOffset(8, 4)
	lab.BackgroundTransparency = 1
	lab.Text = label
	lab.Font = Enum.Font.Gotham
	lab.TextSize = 11
	lab.TextColor3 = COLORS.muted
	lab.TextXAlignment = Enum.TextXAlignment.Left
	local box = Instance.new("TextBox", hold)
	box.Size = UDim2.new(1, -90, 0, 26)
	box.Position = UDim2.fromOffset(8, 24)
	box.BackgroundColor3 = COLORS.panel
	box.Text = tostring(default)
	box.PlaceholderText = minV .. " - " .. maxV
	box.Font = Enum.Font.GothamMedium
	box.TextSize = 13
	box.TextColor3 = COLORS.text
	box.ClearTextOnFocus = false
	corner(box, 6)
	local apply = Instance.new("TextButton", hold)
	apply.Size = UDim2.fromOffset(70, 26)
	apply.Position = UDim2.new(1, -78, 0, 24)
	apply.BackgroundColor3 = COLORS.accent
	apply.Text = "Apply"
	apply.Font = Enum.Font.GothamBold
	apply.TextSize = 12
	apply.TextColor3 = Color3.new(1, 1, 1)
	corner(apply, 6)
	local function doApply()
		local n = tonumber(box.Text)
		if not n then
			notify("Input", "Enter a number")
			return
		end
		n = math.clamp(n, minV, maxV)
		box.Text = tostring(n)
		if onApply then pcall(onApply, n) end
		notify(label, "Set to " .. n)
	end
	apply.MouseButton1Click:Connect(doApply)
	box.FocusLost:Connect(function(enter) if enter then doApply() end end)
end

	local data = S.cfg[slot]
	pcall(function()
		if not data and isfile and isfile("EncryptedHub_MM2_Cfg_" .. slot .. ".json") then
			data = game:GetService("HttpService"):JSONDecode(readfile("EncryptedHub_MM2_Cfg_" .. slot .. ".json"))
			S.cfg[slot] = data
		end
	end)
	if not data then
		notify("Config", "Slot " .. slot .. " empty")
		return
	end
	for k, v in pairs(data) do
		S[k] = v
	end
	if data.speed then S.speedOn = true setSpeed(true) end
	if data.aimbot then S.aimbot = true end
	notify("Config", "Loaded slot " .. slot)

-- drag main
local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = Main.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)
